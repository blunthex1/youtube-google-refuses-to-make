    }
    if (target) {
      video.pause();
      location.href = "/watch?v=" + encodeURIComponent(target.videoId) + "&list=" + encodeURIComponent(list) + "&index=" + (current - 1);
    }
  }
  var state = { key: null, armed: false };
  function onTime(e) {
    if (s.playlist_reverse !== true || !reversed) return;
    const video = e.target;
    if (!(video instanceof HTMLVideoElement) || !video.duration || !isFinite(video.duration)) return;
    if (!/[?&]list=/.test(location.search) || document.querySelector(YT.adShowing) || !video.matches(YT.mainVideo)) return;
    const params = new URLSearchParams(location.search);
    const key = params.get("v") + "|" + params.get("list");
    if (state.key !== key) {
      state.key = key;
      state.armed = false;
    }
    if (e.type === "timeupdate" && video.duration - video.currentTime > 2) state.armed = true;
    if (!state.armed) return;
    if (e.type !== "ended" && video.duration - video.currentTime > 0.6) return;
    state.armed = false;
    goBack(video);
  }
  document.addEventListener("timeupdate", onTime, true);
  document.addEventListener("ended", onTime, true);
  document.addEventListener(YT.navigateFinish, update);

  // src/page/autoplay.js
  var s2 = {};
  function autoplaySettings(next) {
    s2 = next;
  }
  var HELD = "it-autoplay-held";
  var interactedOn = null;
  addEventListener("pointerdown", () => interactedOn = location.href, true);
  addEventListener("keydown", () => interactedOn = location.href, true);
  var youStartedIt = () => interactedOn === location.href;
  document.addEventListener("play", (e) => {
    const v = e.target;
    if (!(v instanceof HTMLVideoElement)) return;
    const player = v.closest(".html5-video-player");
    if (!player || player.classList.contains("ad-showing")) return;
    if (youStartedIt()) {
      player.removeAttribute(HELD);
      return;
    }
    const onVideo = location.pathname === "/watch" && !/[?&]list=/.test(location.search);
    const onChannelHome = CHANNEL_ANY.test(location.pathname) && !CHANNEL_TAB.test(location.pathname);
    if (!(s2.player_autoplay_disable === true && onVideo || s2.channel_trailer_autoplay === false && onChannelHome)) return;
    player.setAttribute(HELD, "");
    try {
      player.pauseVideo();
    } catch {
      v.pause();
    }
    if (!v.paused) v.pause();
  }, true);
  document.addEventListener(YT.navigateStart, () => document.querySelectorAll("[" + HELD + "]").forEach((p) => p.removeAttribute(HELD)));

  // src/page/main.js
  var settings = {};
  var lastVideo = null;
  function guard(name, fn) {
    try {
      fn();
    } catch (e) {
      console.error("[Focus for YouTube] " + name + ":", e);
      try {
        document.dispatchEvent(new CustomEvent("it-error", { detail: JSON.stringify({ name, msg: name + ": " + (e && e.message || e) }) }));
      } catch {
      }
    }
  }
  function resizeSignal() {
    for (const ms of [0, 150, 500, 1200]) setTimeout(() => window.dispatchEvent(new Event("resize")), ms);
  }
  document.addEventListener("it-settings", (e) => {
    let next;
    try {
      next = JSON.parse(e.detail);
    } catch {
      return;
    }
    const first = !Object.keys(settings).length;
    settings = next;
    guard("autoplay", () => autoplaySettings(settings));
    guard("reverse playlist", () => playlistSettings(settings));
    guard("theater mode", forcedTheaterFallback);
    guard("player size", resizeSignal);
    if (first) {
      guard("blank video watch", watchForBlankVideo);
      guard("new video", onNewVideo);
    }
    guard("fit recheck", watchFitSize);
    if (!first) {
      guard("quality", applyQuality);
      guard("audio enhancements", () => {
        const p = document.querySelector(YT.player);
        if (p) {
          delete p.dataset.itEnh;
          enhancements(p, settings);
        }
      });
    }
  });
  var fitTimer = null;
  function watchFitSize() {
    clearInterval(fitTimer);
    if (settings.fit_recheck !== true || settings.player_size !== "fit_to_window" || location.pathname !== "/watch") return;
    let n = 0, lastNudge = 0;
    fitTimer = setInterval(() => {
      if (++n > 200) return clearInterval(fitTimer);
      const flexy = document.querySelector(YT.watchFlexy), v = document.querySelector(YT.playerVideo);
      const box = document.querySelector(YT.fullBleedBox);
      if (!flexy || !v || !box || !isTheater(flexy) || flexy.hasAttribute("fullscreen")) return;
      const b = box.getBoundingClientRect(), r = v.getBoundingClientRect();
      if (b.height > 100 && r.height > 2 && r.height < b.height - 4 && Date.now() - lastNudge > 600) {
        lastNudge = Date.now();
        window.dispatchEvent(new Event("resize"));
      }
    }, 200);
  }
  var rescueTimer = null;
  function watchForBlankVideo() {
    clearInterval(rescueTimer);
    document.documentElement.removeAttribute("it-video-rescue");
    if (location.pathname !== "/watch") return;
    let n = 0;
    rescueTimer = setInterval(() => {
      const p = document.querySelector(YT.player), v = document.querySelector(YT.playerVideo);
      if (!p || !v) {
        if (++n > 20) clearInterval(rescueTimer);
        return;
      }
      const pr = p.getBoundingClientRect(), vr = v.getBoundingClientRect();
      if (pr.width > 100 && pr.height > 100 && (vr.width < 2 || vr.height < 2)) {
        document.documentElement.setAttribute("it-video-rescue", "");
        resizeSignal();
      }
      if (++n > 20) clearInterval(rescueTimer);
    }, 700);
  }
  document.dispatchEvent(new CustomEvent("it-settings-request"));
  function whenPlayerReady(cb, tries = 0) {
    const player = document.querySelector(YT.player);
    if (player && typeof player.getAvailableQualityLevels === "function") return cb(player);
    if (tries < 50) setTimeout(() => whenPlayerReady(cb, tries + 1), 100);
  }
  function onNewVideo() {
    if (location.pathname !== "/watch" || !Object.keys(settings).length) return;
    if (lastVideo === location.href) return;
    lastVideo = location.href;
    whenPlayerReady((player) => {
      delete player.dataset.itQuality;
      guard("quality", applyQuality);
      guard("subtitles", () => subtitles(player, settings));
      guard("auto-dubbing", () => disableAutoDubbing(player, settings));
      guard("up next autoplay", () => upNextAutoplay(settings));
      guard("audio enhancements", () => enhancements(player, settings));
      guard("player size", resizeSignal);
    });
  }
  function applyQuality() {
    const player = document.querySelector(YT.player);
    if (player) setQuality(player, wantedQuality(settings));
  }
  document.addEventListener(YT.navigateFinish, () => {
    guard("blank video watch", watchForBlankVideo);
    guard("fit recheck", watchFitSize);
    guard("new video", onNewVideo);
    guard("theater mode", forcedTheaterFallback);
  });
  document.addEventListener("loadedmetadata", (e) => {
    if (e.target.closest?.(YT.player)) applyQuality();
  }, true);
  document.addEventListener("fullscreenchange", () => setTimeout(applyQuality, 300));
  document.addEventListener("visibilitychange", () => guard("quality", applyQuality));
  var theaterWait = null;
  function forcedTheaterFallback() {
    if (settings.forced_theater_mode !== true || location.pathname !== "/watch") return;
    if (theaterWait) return;
    const started = Date.now();
    let lastClick = 0;
    const step = () => {
      if (settings.forced_theater_mode !== true || location.pathname !== "/watch" || Date.now() - started > 3e4) return stop();
      const flexy = document.querySelector(YT.watchFlexy);
      if (flexy && isTheater(flexy)) {
        resizeSignal();
        return stop();
      }
      const button = document.querySelector(YT.theaterButton);
      if (flexy && button && Date.now() - lastClick > 700) {
        lastClick = Date.now();
        button.click();
      }
    };
    theaterWait = setInterval(step, 200);
    step();
    function stop() {
      clearInterval(theaterWait);
      theaterWait = null;
    }
  }
})();
